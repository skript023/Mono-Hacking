cmake_minimum_required(VERSION 3.20)
project(quantum_ui VERSION 0.1.0 LANGUAGES CXX)
include(FetchContent)
set(QUANTUM_UI_IMGUI_TARGET imgui CACHE STRING "Existing Dear ImGui target")
set(QUANTUM_UI_BACKENDS "" CACHE STRING "Optional backend targets: dx9;dx11;dx12;vulkan;opengl")
option(QUANTUM_UI_BUILD_TESTS "Build standalone UI regression tests" OFF)

if(NOT TARGET ${QUANTUM_UI_IMGUI_TARGET})
    FetchContent_Declare(imgui
        GIT_REPOSITORY https://github.com/ocornut/imgui.git
        GIT_TAG 725d185a31f860b74014aba167aa69003d12c01a)
    FetchContent_MakeAvailable(imgui)
    add_library(${QUANTUM_UI_IMGUI_TARGET} STATIC
        ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
        ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp)
    target_include_directories(${QUANTUM_UI_IMGUI_TARGET} PUBLIC ${imgui_SOURCE_DIR})
endif()

add_library(quantum_ui_core INTERFACE)
add_library(quantum_ui::core ALIAS quantum_ui_core)
target_include_directories(quantum_ui_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(quantum_ui_core INTERFACE cxx_std_17)
add_library(quantum_ui STATIC src/theme.cpp src/widgets.cpp src/window.cpp src/list.cpp src/menu.cpp)
add_library(quantum_ui::ui ALIAS quantum_ui)
target_link_libraries(quantum_ui PUBLIC quantum_ui_core ${QUANTUM_UI_IMGUI_TARGET})

# Explicit opt-in: a host already compiling ImGui backends must not compile them twice.
if(QUANTUM_UI_BACKENDS)
    if(NOT WIN32)
        message(FATAL_ERROR "Packaged platform backends currently require Windows")
    endif()
    set(QUANTUM_UI_IMGUI_SOURCE_DIR "${imgui_SOURCE_DIR}" CACHE PATH "ImGui source containing backends/")
    if(NOT EXISTS "${QUANTUM_UI_IMGUI_SOURCE_DIR}/backends/imgui_impl_win32.cpp")
        message(FATAL_ERROR "Set QUANTUM_UI_IMGUI_SOURCE_DIR to the matching ImGui sources")
    endif()
    add_library(quantum_ui_win32 STATIC "${QUANTUM_UI_IMGUI_SOURCE_DIR}/backends/imgui_impl_win32.cpp")
    add_library(quantum_ui::win32 ALIAS quantum_ui_win32)
    target_link_libraries(quantum_ui_win32 PUBLIC ${QUANTUM_UI_IMGUI_TARGET} PRIVATE dwmapi)
    target_include_directories(quantum_ui_win32 PUBLIC "${QUANTUM_UI_IMGUI_SOURCE_DIR}/backends")
    foreach(api IN LISTS QUANTUM_UI_BACKENDS)
        if(NOT api MATCHES "^(dx9|dx11|dx12|vulkan|opengl)$")
            message(FATAL_ERROR "Unknown UI backend: ${api}")
        endif()
        set(backend ${api})
        if(api STREQUAL "opengl")
            set(backend opengl3)
        endif()
        add_library(quantum_ui_${api} STATIC "${QUANTUM_UI_IMGUI_SOURCE_DIR}/backends/imgui_impl_${backend}.cpp")
        add_library(quantum_ui::${api} ALIAS quantum_ui_${api})
        target_include_directories(quantum_ui_${api} PUBLIC "${QUANTUM_UI_IMGUI_SOURCE_DIR}/backends")
        target_link_libraries(quantum_ui_${api} PUBLIC quantum_ui quantum_ui_win32)
        if(api STREQUAL "dx9")
            target_link_libraries(quantum_ui_${api} PRIVATE d3d9)
        elseif(api STREQUAL "dx11")
            target_link_libraries(quantum_ui_${api} PRIVATE d3d11 dxgi d3dcompiler)
        elseif(api STREQUAL "dx12")
            target_link_libraries(quantum_ui_${api} PRIVATE d3d12 dxgi d3dcompiler)
        elseif(api STREQUAL "opengl")
            target_link_libraries(quantum_ui_${api} PRIVATE opengl32)
        elseif(api STREQUAL "vulkan")
            if(NOT TARGET Vulkan::Headers)
                FetchContent_Declare(VulkanHeaders GIT_REPOSITORY https://github.com/KhronosGroup/Vulkan-Headers.git GIT_TAG v1.3.290)
                set(VULKAN_HEADERS_ENABLE_MODULE OFF CACHE BOOL "Build Vulkan C++ module")
                FetchContent_MakeAvailable(VulkanHeaders)
            endif()
            target_link_libraries(quantum_ui_${api} PUBLIC Vulkan::Headers)
            target_compile_definitions(quantum_ui_${api} PUBLIC IMGUI_IMPL_VULKAN_NO_PROTOTYPES)
        endif()
    endforeach()
endif()
if(QUANTUM_UI_BUILD_TESTS)
    enable_testing()
    add_executable(quantum_ui_tests tests/navigation.cpp)
    target_link_libraries(quantum_ui_tests PRIVATE quantum_ui)
    add_test(NAME quantum_ui_regression COMMAND quantum_ui_tests)
endif()

option(QUANTUM_UI_BUILD_EXAMPLE "Build a standalone DX11 window demo" OFF)
if(QUANTUM_UI_BUILD_EXAMPLE)
    if(NOT TARGET quantum_ui_dx11)
        message(FATAL_ERROR "The demo requires dx11 in QUANTUM_UI_BACKENDS")
    endif()
    add_executable(quantum_ui_demo examples/win32_dx11.cpp)
    target_link_libraries(quantum_ui_demo PRIVATE quantum_ui_dx11)
endif()
